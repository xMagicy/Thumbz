10. Pool gezondheid metrics
Run deze queries en geef output:

-- Aantal actieve thumbnails per category
SELECT app_category, COUNT(*) FROM thumbnails WHERE archived IS NOT true GROUP BY 1 ORDER BY 2 DESC;

-- Aantal Shorts/trailers nog in pool (mag 0 zijn)
SELECT COUNT(*) FROM thumbnails 
WHERE archived IS NOT true 
  AND (title ~* '\b(short|trailer|teaser)s?\b' OR channel_name ~* '\b(studios|pictures|films)\b');

-- Channel diversity check
SELECT channel_name, COUNT(*) FROM thumbnails 
WHERE archived IS NOT true 
GROUP BY 1 HAVING COUNT(*) > 2;

-- ELO distributie sanity
SELECT 
  COUNT(*) total,
  ROUND(AVG(elo_rating)) avg_elo,
  MIN(elo_rating), MAX(elo_rating),
  COUNT(*) FILTER (WHERE battle_count = 0) untested,
  COUNT(*) FILTER (WHERE is_user_upload = true) user_uploads
FROM thumbnails WHERE archived IS NOT true;

-- Top 10 op breakout_score (test of het veld nu werkt)
SELECT title, channel_name, breakout_score, view_count, subscriber_count
FROM thumbnails 
WHERE archived IS NOT true AND breakout_score IS NOT NULL
ORDER BY breakout_score DESC LIMIT 10;
