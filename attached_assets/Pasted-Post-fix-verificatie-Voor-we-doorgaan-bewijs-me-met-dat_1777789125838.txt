1. Database state
Run deze queries in deze exacte volgorde, plak ALLE output:

-- 1.1 Pool gezondheid
SELECT 
  COUNT(*) FILTER (WHERE archived = false) AS active_total,
  COUNT(*) FILTER (WHERE archived = false AND source = 'youtube') AS active_youtube,
  COUNT(*) FILTER (WHERE archived = false AND source = 'user') AS active_user_uploads,
  COUNT(*) FILTER (WHERE archived = true) AS archived_total
FROM thumbnails;

-- 1.2 Shorts leakage check (moet 0 zijn)
SELECT id, title, channel_name, archived, status
FROM thumbnails 
WHERE archived = false 
  AND title ~* '#?(shorts?|reels?|ytshorts?|minivlog|tiktoks?)\b';

-- 1.3 De twee specifieke cases die ik live zag
SELECT id, title, channel_name, archived, status, niche, app_category
FROM thumbnails
WHERE title ILIKE '%adult money haul%'
   OR title ILIKE '%soap with ashes%'
   OR channel_name ILIKE '%creative explained%'
   OR channel_name ILIKE '%swara singh%';

-- 1.4 Per-category breakdown na cleanup
SELECT app_category, COUNT(*) FILTER (WHERE archived = false) AS active
FROM thumbnails 
WHERE source = 'youtube'
GROUP BY app_category 
ORDER BY 2 DESC;

-- 1.5 Hoeveel rijen heeft de cleanup geraakt vandaag
SELECT COUNT(*) 
FROM thumbnails 
WHERE status = 'shorts_archived' 
   OR (archived = true AND title ~* '#?(shorts?|reels?)\b');
